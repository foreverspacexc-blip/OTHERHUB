	)

	local button = Instance.new("TextButton")
	button.Name = name .. "_Button"
	button.Size = UDim2.new(0, 78, 0, 26)
	button.Position = UDim2.new(1, -88, 0.5, -13)
	button.BackgroundColor3 = RED
	button.Text = "เปิด"
	button.TextColor3 = WHITE
	button.TextSize = 11
	button.Font = Enum.Font.GothamMedium
	button.AutoButtonColor = false
	button.Parent = row

	Corner(button, 6)

	local busy = false

	button.MouseButton1Click:Connect(function()
		if busy then
			return
		end

		busy = true
		button.Text = "..."

		local ok = pcall(function()
			loadstring(game:HttpGet(url))()
		end)

		if ok then
			button.Text = "เปิดแล้ว ✓"
			button.BackgroundColor3 = GREEN
		else
			button.Text = "ผิดพลาด"
		end

		task.delay(1.2, function()
			if button.Parent then
				button.Text = "เปิด"
				button.BackgroundColor3 = RED
			end
			busy = false
		end)
	end)

	return row
end

local NoclipActive = false
local NoclipConn   = nil

local function EnableNoclip()
	NoclipActive = true
	NoclipConn = RunService.Stepped:Connect(function()
		local char = Player.Character
		if not char then return end
		for _, p in pairs(char:GetDescendants()) do
			if p:IsA("BasePart") then
				p.CanCollide = false
			end
		end
	end)
end

local function DisableNoclip()
	NoclipActive = false
	if NoclipConn then
		NoclipConn:Disconnect()
		NoclipConn = nil
	end
	local char = Player.Character
	if not char then return end
	for _, p in pairs(char:GetDescendants()) do
		if p:IsA("BasePart") then
			p.CanCollide = true
		end
	end
end

	InfJumpActive = true

TPToolActive  = false
local TPTool  = nil
local TPRespawnConn = nil

function CreateTPTool()
	local bp   = Player.Backpack
	local char = Player.Character
	if not bp or not char then return end

	for _, v in pairs(bp:GetChildren()) do
		if v.Name == "TP NEXORA" then v:Destroy() end
	end
	for _, v in pairs(char:GetChildren()) do
		if v.Name == "TP NEXORA" then v:Destroy() end
	end

	local tool = Instance.new("Tool")
	tool.Name           = "TP NEXORA"
	tool.RequiresHandle = false
	tool.CanBeDropped   = false
	tool.ToolTip        = "ถือแล้วกดตรงไหนในแมพ → วาร์ปทันที"
	tool.Parent         = bp

	local handle = Instance.new("Part")
	handle.Name            = "Handle"
	handle.Size            = Vector3.new(0.1, 0.1, 0.1)
	handle.Transparency    = 1
	handle.CanCollide      = false
	handle.Massless        = true
	handle.Parent          = tool

	tool.Activated:Connect(function()
		local mouse   = Player:GetMouse()
		local camPart = workspace.CurrentCamera
		if not camPart then return end

		local unitRay = camPart:ScreenPointToRay(mouse.X, mouse.Y)
		local rp      = RaycastParams.new()
		rp.FilterType = Enum.RaycastFilterType.Exclude
		rp.FilterDescendantsInstances = { Player.Character }

		local result = workspace:Raycast(unitRay.Origin, unitRay.Direction * 3000, rp)
		if not result then return end

		local landPos = result.Position + Vector3.new(0, 3.5, 0)

		local c2  = Player.Character
		local hrp = c2 and c2:FindFirstChild("HumanoidRootPart")
		if not hrp then return end

		hrp.CFrame = CFrame.new(landPos)

		local flash = Instance.new("SelectionBox")
		flash.Color3     = Color3.fromRGB(255, 25, 45)
		flash.SurfaceColor3 = Color3.fromRGB(255, 25, 45)
		flash.LineThickness  = 0.04
		flash.SurfaceTransparency = 0.7
		flash.Adornee    = hrp
		flash.Parent     = hrp
		TweenService:Create(flash, TweenInfo.new(0.35), { SurfaceTransparency = 1, LineThickness = 0 }):Play()
		game:GetService("Debris"):AddItem(flash, 0.4)
	end)

	TPTool = tool
end

local function EnableTPTool()
	TPToolActive = true
	CreateTPTool()
	if TPRespawnConn then TPRespawnConn:Disconnect() end
	TPRespawnConn = Player.CharacterAdded:Connect(function()
		task.wait(0.8)
		if TPToolActive then CreateTPTool() end
	end)
end

local function DisableTPTool()
	TPToolActive = false
	if TPRespawnConn then TPRespawnConn:Disconnect(); TPRespawnConn = nil end
	local bp = Player.Backpack
	if bp then
		for _, v in pairs(bp:GetChildren()) do
			if v.Name == "TP NEXORA" then v:Destroy() end
		end
	end
	local char = Player.Character
	if char then
		for _, v in pairs(char:GetChildren()) do
			if v.Name == "TP NEXORA" then v:Destroy() end
		end
	end
	TPTool = nil
end

local function UpdateHP()
	local character = Player.Character
	if not character then return end

	local humanoid = character:FindFirstChildOfClass("Humanoid")
	if not humanoid then return end

	local hp = math.floor(humanoid.Health + 0.5)
	local maxHP = math.floor(humanoid.MaxHealth + 0.5)

	HPText.Text = hp .. "/" .. maxHP

	local percent = 0
	if maxHP > 0 then
		percent = math.clamp(humanoid.Health / humanoid.MaxHealth, 0, 1)
	end

	HPBar.Size = UDim2.new(percent, 0, 1, 0)
end

RunService.Heartbeat:Connect(UpdateHP)

Player.CharacterAdded:Connect(function()
	task.wait(1)
	UpdateHP()
end)

local Frames = 0
local LastFPS = os.clock()

RunService.RenderStepped:Connect(function()
	Frames += 1

	local now = os.clock()

	if now - LastFPS >= 0.5 then
		local fps = math.floor(Frames / (now - LastFPS) + 0.5)

		FPSText.Text = tostring(fps)
		FPSBar.Size = UDim2.new(math.clamp(fps / 120, 0, 1), 0, 1, 0)

		Frames = 0
		LastFPS = now
	end
end)

-- 3. Walk Fling  (velocity spike from IY source)
local _flingActive, _flingConn, _flingNoclipConn = false, nil, nil
local _flingOrigSpeed = 16
local function EnableWalkFling()
	_flingActive = true
	local c = GetChar(); local h = GetHum(c); local hrp = GetHRP(c)
	if not h or not hrp then return end
	_flingOrigSpeed = h.WalkSpeed
	-- noclip so we don't get stopped by walls
	_flingNoclipConn = RunService.Stepped:Connect(function()
		local ch = GetChar(); if not ch then return end
		for _, p in pairs(ch:GetDescendants()) do
			if p:IsA("BasePart") then p.CanCollide = false end
		end
	end)
	_flingConn = RunService.Heartbeat:Connect(function()
		xpcall(function()
			local ch = GetChar(); local root = GetHRP(ch)
			if not root then return end
			local vel = root.AssemblyLinearVelocity
			root.AssemblyLinearVelocity = vel * 10000 + Vector3.new(0, 10000, 0)
			RunService.RenderStepped:Wait()
			if root and root.Parent then root.AssemblyLinearVelocity = vel end
		end, function() end)
	end)
end
local function DisableWalkFling()
	_flingActive = false
	if _flingConn then _flingConn:Disconnect(); _flingConn = nil end
	if _flingNoclipConn then _flingNoclipConn:Disconnect(); _flingNoclipConn = nil end
	local c = GetChar(); local h = GetHum(c)
	if h then h.WalkSpeed = _flingOrigSpeed end
	if c then
		for _, p in pairs(c:GetDescendants()) do
			if p:IsA("BasePart") then pcall(function() p.CanCollide = true end) end
		end
	end
end

	_invisActive = true

						no.Parent = char

		char:MoveTo(box.Position + Vector3.new(0, 1, 0))

			pcall(function() conn:Disconnect(); box:Destroy() end)

-- 7. Server Hop
local function DoServerHop()
	task.spawn(function()
		xpcall(function()
			local req = game:HttpGet("https://games.roblox.com/v1/games/" .. game.PlaceId .. "/servers/Public?sortOrder=Desc&limit=100&excludeFullGames=true")
			local body = HttpService:JSONDecode(req)
			local servers = {}
			if body and body.data then
				for _, v in pairs(body.data) do
					if type(v) == "table" and v.id and v.id ~= game.JobId then
						table.insert(servers, v.id)
					end
				end
			end
			if #servers > 0 then
				TeleportService:TeleportToPlaceInstance(game.PlaceId, servers[math.random(1, #servers)], Player)
			else
				TeleportService:Teleport(game.PlaceId, Player)
			end
		end, function(e)
			pcall(function() TeleportService:Teleport(game.PlaceId, Player) end)
		end)
	end)
end

local function EnableHeadless()
	_headlessOn = true
	xpcall(function()
		local c = GetChar(); if not c then return end
		local head = c:FindFirstChild("Head")
		if head then
			head.Transparency = 1
			for _, d in pairs(head:GetChildren()) do

-- 15. Play Animation ID
local _currentAnimTrack = nil
local function PlayAnimId(id)
	xpcall(function()
		local c = GetChar(); local h = GetHum(c)
		if not h then return end
		local numId = tostring(id):match("%d+")
		if not numId then return end
		if _currentAnimTrack then pcall(function() _currentAnimTrack:Stop() end) end
		local animator = h:FindFirstChildOfClass("Animator")
		if not animator then animator = Instance.new("Animator"); animator.Parent = h end
		local anim = Instance.new("Animation")
		anim.AnimationId = "rbxassetid://" .. numId
		local track = animator:LoadAnimation(anim)
		track.Priority = Enum.AnimationPriority.Action
		track:Play()
		_currentAnimTrack = track
	end, function() end)
end
local function StopAnim()
	if _currentAnimTrack then pcall(function() _currentAnimTrack:Stop() end); _currentAnimTrack = nil end
end

local Closing = false

local function OpenGUI()
	if Main.Visible then
		return
	end

	Main.Visible = true
	Main.AnchorPoint = Vector2.new(0.5, 0.5)
	Main.Position = UDim2.new(0.5, 0, 0.5, 0)
	Main.Size = UDim2.new(0, 600, 0, 430)

	TweenService:Create(
		Main,
		TweenInfo.new(0.3, Enum.EasingStyle.Back, Enum.EasingDirection.Out),
		{ Size = UDim2.new(0, 680, 0, 490) }
	):Play()
end

local function CloseGUI()
	if not Main.Visible or Closing then
		return
	end

	Closing = true

	local tween = TweenService:Create(
		Main,
		TweenInfo.new(0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.In),
		{ Size = UDim2.new(0, 600, 0, 430) }
	)

	tween:Play()
	tween.Completed:Wait()

	Main.Visible = false
	Main.Size = UDim2.new(0, 680, 0, 490)

	Closing = false
end

--==================================================
-- LOGO DRAG + CLICK
--==================================================

local LogoDragging = false
local LogoMoved = false
local LogoDragStart = nil
local LogoStartPos = nil

local DRAG_THRESHOLD = 8

LogoButton.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1
		or input.UserInputType == Enum.UserInputType.Touch then

		LogoDragging = true
		LogoMoved = false

		LogoDragStart = input.Position
		LogoStartPos = LogoButton.Position

		TweenService:Create(
			LogoButton,
			TweenInfo.new(0.1, Enum.EasingStyle.Quad),
			{ Size = UDim2.new(0, 64, 0, 64) }
		):Play()
	end
end)

UserInputService.InputChanged:Connect(function(input)
	if not LogoDragging then
		return
	end

	if input.UserInputType ~= Enum.UserInputType.MouseMovement
		and input.UserInputType ~= Enum.UserInputType.Touch then
		return
	end

	local delta = input.Position - LogoDragStart

	if math.abs(delta.X) >= DRAG_THRESHOLD
		or math.abs(delta.Y) >= DRAG_THRESHOLD then
		LogoMoved = true
	end

	LogoButton.Position = UDim2.new(
		LogoStartPos.X.Scale,
		LogoStartPos.X.Offset + delta.X,
		LogoStartPos.Y.Scale,
		LogoStartPos.Y.Offset + delta.Y
	)
end)

UserInputService.InputEnded:Connect(function(input)
	if input.UserInputType ~= Enum.UserInputType.MouseButton1
		and input.UserInputType ~= Enum.UserInputType.Touch then
		return
	end

	if not LogoDragging then
		return
	end

	LogoDragging = false

	TweenService:Create(
		LogoButton,
		TweenInfo.new(0.12, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
		{ Size = UDim2.new(0, 58, 0, 58) }
	):Play()

	if not LogoMoved then
		if Main.Visible then
			task.spawn(CloseGUI)
		else
			OpenGUI()
		end
	end
end)
