local function GetRandomName()
    if not Config.AntiDetection then return "Component" end
    local chars = "abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789"
    local length = math.random(10, 20)
    local randomString = ""
    for i = 1, length do
        local rand = math.random(1, #chars)
        randomString = randomString .. string.sub(chars, rand, rand)
    end
    return randomString
end

function Library:SaveFlags()
    local success, err = pcall(function()
        local data = HttpService:JSONEncode(Library.Flags)
        writefile(string.format("Invis_Universal/%s.json", tostring(game.GameId)), data)
    end)
end

function Library:LoadFlags()
    local path = string.format("Invis_Universal/%s.json", tostring(game.GameId))
    if isfile(path) then
        local success, data = pcall(function() return readfile(path) end)
        if success and data then
            local decodeSuccess, decoded = pcall(function() return HttpService:JSONDecode(data) end)
            if decodeSuccess and decoded then Library.Flags = decoded end
        end
    end
end

    if Library.Flags["chat_commands_toggle_flag"] == false then return end

function Library:MakeDraggable(clickFrame, dragFrame)
    local dragging = false
    local dragInput, dragStart, startPos

    clickFrame.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = dragFrame.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then dragging = false end
            end)
        end
    end)

    clickFrame.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if dragging and input == dragInput then
            local delta = input.Position - dragStart
            dragFrame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)
end

function Library:AnimateWindow(frame, open)
    frame.Visible = true
    if open then
        frame.Size = UDim2.new(0, 0, 0, 0)
        TweenService:Create(frame, TweenInfo.new(0.5, Enum.EasingStyle.Circular, Enum.EasingDirection.Out), {
            Size = UDim2.new(0, 650, 0, 400)
        }):Play()
    else
        local closeTween = TweenService:Create(frame, TweenInfo.new(0.5, Enum.EasingStyle.Circular, Enum.EasingDirection.In), {
            Size = UDim2.new(0, 0, 0, 0)
        })
        closeTween:Play()
        closeTween.Completed:Once(function()
            if not Library.Enabled then frame.Visible = false end
        end)
    end
end

            end
            if (CONTROL.L + CONTROL.R) ~= 0 or (CONTROL.F + CONTROL.B) ~= 0 or (CONTROL.Q + CONTROL.E) ~= 0 then
                BodyVelocityInstance.Velocity = ((camera.CFrame.LookVector * (CONTROL.F + CONTROL.B)) + ((camera.CFrame * CFrame.new(CONTROL.L + CONTROL.R, (CONTROL.F + CONTROL.B + CONTROL.Q + CONTROL.E) * 0.2, 0).p) - camera.CFrame.p)) * SPEED
                lCONTROL = {F = CONTROL.F, B = CONTROL.B, L = CONTROL.L, R = CONTROL.R}
            elseif (CONTROL.L + CONTROL.R) == 0 and (CONTROL.F + CONTROL.B) == 0 and (CONTROL.Q + CONTROL.E) == 0 and SPEED ~= 0 then
                BodyVelocityInstance.Velocity = ((camera.CFrame.LookVector * (lCONTROL.F + lCONTROL.B)) + ((camera.CFrame * CFrame.new(lCONTROL.L + lCONTROL.R, (lCONTROL.F + lCONTROL.B + CONTROL.Q + CONTROL.E) * 0.2, 0).p) - camera.CFrame.p)) * SPEED
            else
                BodyVelocityInstance.Velocity = Vector3.new(0, 0, 0)
            end
            BodyGyroInstance.CFrame = camera.CFrame
        until not FlyState
        if BodyGyroInstance then BodyGyroInstance:Destroy() end
        if BodyVelocityInstance then BodyVelocityInstance:Destroy() end
        if humanoid then humanoid.PlatformStand = false end
    end)

    flyKeyDown = UserInputService.InputBegan:Connect(function(input, processed)
        if processed then return end
        -- นำค่าความเร็ว FlySpeed จากตัวแปรหลักมาใช้และปรับตัวคูณให้สัมพันธ์กับสปีดดั้งเดิม (IY speed 1 = FlySpeed 20)
        local convertedSpeed = FlySpeed / 20 
        if input.KeyCode == Enum.KeyCode.W then CONTROL.F = convertedSpeed
        elseif input.KeyCode == Enum.KeyCode.S then CONTROL.B = -convertedSpeed
        elseif input.KeyCode == Enum.KeyCode.A then CONTROL.L = -convertedSpeed
        elseif input.KeyCode == Enum.KeyCode.D then CONTROL.R = convertedSpeed
        elseif input.KeyCode == Enum.KeyCode.E then CONTROL.Q = convertedSpeed * 2
        elseif input.KeyCode == Enum.KeyCode.Q then CONTROL.E = -convertedSpeed * 2
        end
        pcall(function() camera.CameraType = Enum.CameraType.Track end)
    end)

    flyKeyUp = UserInputService.InputEnded:Connect(function(input, processed)
        if processed then return end
        if input.KeyCode == Enum.KeyCode.W then CONTROL.F = 0
        elseif input.KeyCode == Enum.KeyCode.S then CONTROL.B = 0
        elseif input.KeyCode == Enum.KeyCode.A then CONTROL.L = 0
        elseif input.KeyCode == Enum.KeyCode.D then CONTROL.R = 0
        elseif input.KeyCode == Enum.KeyCode.E then CONTROL.Q = 0
        elseif input.KeyCode == Enum.KeyCode.Q then CONTROL.E = 0
        end
    end)
end

function mobilefly()
    NOFLY()
    
    local char = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
    local root = char:WaitForChild("HumanoidRootPart", 10)
    local humanoid = char:FindFirstChildOfClass("Humanoid")
    local camera = workspace.CurrentCamera
    local v3inf = Vector3.new(9e9, 9e9, 9e9)
    local v3zero = Vector3.new(0, 0, 0)
    
    if not root or not humanoid then 
        print("Error: Character parts missing for mobile flight")
        return 
    end
    
    BodyVelocityInstance = Instance.new("BodyVelocity")
    BodyVelocityInstance.Name = velocityHandlerName
    BodyVelocityInstance.Parent = root
    BodyVelocityInstance.MaxForce = v3inf
    BodyVelocityInstance.Velocity = v3zero

    BodyGyroInstance.Name = gyroHandlerName
    BodyGyroInstance.Parent = root
    BodyGyroInstance.MaxTorque = v3inf
    BodyGyroInstance.P = 9e4
    
    mfly1 = LocalPlayer.CharacterAdded:Connect(function(newChar)
        root = newChar:WaitForChild("HumanoidRootPart", 10)
        humanoid = newChar:WaitForChild("Humanoid", 10)
        
        if root then
            BodyVelocityInstance = Instance.new("BodyVelocity")
            BodyVelocityInstance.Name = velocityHandlerName
            BodyVelocityInstance.Parent = root
            BodyVelocityInstance.MaxForce = v3inf
            BodyVelocityInstance.Velocity = v3zero
            
            BodyGyroInstance = Instance.new("BodyGyro")
            BodyGyroInstance.Name = gyroHandlerName
            BodyGyroInstance.Parent = root
            BodyGyroInstance.MaxTorque = v3inf
            BodyGyroInstance.P = 9e4
        else
            print("Error: Failed to re-bind root part on respawn")
        end
    end)
    
    mfly2 = game:GetService("RunService").RenderStepped:Connect(function()
        if LocalPlayer.Character and root and humanoid and BodyVelocityInstance and BodyGyroInstance then
            humanoid.PlatformStand = true
            BodyGyroInstance.CFrame = camera.CFrame
            
            local moveVector = v3zero
            local controlModule = nil
            
            pcall(function()
                controlModule = require(LocalPlayer.PlayerScripts:FindFirstChild("PlayerModule"):FindFirstChild("ControlModule"))
            end)
            
            if controlModule and controlModule.GetMoveVector then
                local rawVector = controlModule:GetMoveVector()
                moveVector = Vector3.new(rawVector.X, 0, rawVector.Z)
            else
                moveVector = humanoid.MoveDirection
            end
            
            local currentSpeed = tonumber(FlySpeed) or 50
            local direction = camera.CFrame:VectorToWorldSpace(moveVector)
            BodyVelocityInstance.Velocity = direction * currentSpeed
        else
            print("Error: Mobile flight loop encountered missing instances")
        end
    end)
end

                    NoclipCache[part] = part.CanCollide
                end

            for part, originalCollide in pairs(NoclipCache) do

local function ToggleAimbot(state)
    AimbotState = state
    if AimbotConnection then AimbotConnection:Disconnect() AimbotConnection = nil end
    if AimbotState then
        AimbotConnection = game:GetService("RunService").RenderStepped:Connect(function()
            local camera = workspace.CurrentCamera
            local closestPlayer = nil
            local shortestDistance = math.huge
            
            for _, player in ipairs(game:GetService("Players"):GetPlayers()) do
                if player ~= LocalPlayer and player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
                    local humanoid = player.Character:FindFirstChildWhichIsA("Humanoid")
                    if humanoid and humanoid.Health > 0 then
                        local screenPos, onScreen = camera:WorldToViewportPoint(player.Character.HumanoidRootPart.Position)
                        if onScreen then
                            local mousePos = UserInputService:GetMouseLocation()
                            local distance = (Vector2.new(screenPos.X, screenPos.Y) - mousePos).Magnitude
                            if distance < shortestDistance then
                                closestPlayer = player
                                shortestDistance = distance
                            end
                        end
                    end
                end
            end
            
            if closestPlayer and closestPlayer.Character and closestPlayer.Character:FindFirstChild("HumanoidRootPart") then
                camera.CFrame = CFrame.new(camera.CFrame.Position, closestPlayer.Character.HumanoidRootPart.Position)
            end
        end)
    end

local function IsSafeToKill(targetPlayer)
    if not targetPlayer or player == LocalPlayer or not targetPlayer.Character then return false end
    
    local root = targetPlayer.Character:FindFirstChild("HumanoidRootPart")
    local humanoid = targetPlayer.Character:FindFirstChildWhichIsA("Humanoid")
    
    if not root or not humanoid then return false end
    if humanoid.Health <= 0 then return false end -- ป้องกันคนตายอยู่
    if targetPlayer.Character:FindFirstChildWhichIsA("ForceField") then return false end -- ป้องกันคนมี ForceField
    
    if root.Position.Y <= (workspace.FallenPartsDestroyHeight + 20) then return false end
    
    return true
end

local function StartSwordLoopKill(targetName)
    if not HasSwordEquipped() then return end
    
    local myRoot = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not myRoot then return end
    
    SavedPosition = myRoot.CFrame
    LoopKillState = true
    
    task.spawn(function()
        while LoopKillState and task.wait() do
            -- ตรวจสอบเงื่อนไขความปลอดภัย หากปลดการถือดาบให้หยุดการทำงานและวาร์ปกลับทันที
            if not HasSwordEquipped() then 
                break 
            end
            
            local target = game:GetService("Players"):FindFirstChild(targetName)
            if target and IsSafeToKill(target) then
                local targetRoot = target.Character.HumanoidRootPart
                -- ทำการวาร์ปดาบไปสับตำแหน่งหลังเป้าหมายต่อเนื่อง
                myRoot.CFrame = targetRoot.CFrame * CFrame.new(0, 0, 1.5)
            else
                break
            end
        end
        
        LoopKillState = false
        if SavedPosition and myRoot then
            myRoot.CFrame = SavedPosition
            SavedPosition = nil
        end
    end)
end

function UpdateComponentSearch()
    local searchName = tostring(Library.Flags["search_box_flag"] or ""):lower()
    local searchType = tostring(Library.Flags["search_type_dropdown_flag"] or "Choose Type"):lower()
    
    for _, page in pairs(ContentContainer:GetChildren()) do
        if page:IsA("Frame") then
            for _, section in pairs(page:GetChildren()) do
                if section:IsA("ScrollingFrame") then
                    for _, element in pairs(section:GetChildren()) do
                        if element:IsA("Frame") or element:IsA("TextButton") then
                            local eName = tostring(element:FindFirstChildOfClass("TextLabel") and element:FindFirstChildOfClass("TextLabel").Text or ""):lower()
                            if eName == "" and element:IsA("TextButton") then
                                eName = tostring(element.Text):lower()
                            end
                            
                            local isMatchedName = (searchName == "" or string.find(eName, searchName))
                            local isMatchedType = true
                            
                            if searchType ~= "choose type" and searchType ~= "" then
                                if searchType == "toggle" then
                                    isMatchedType = (element:IsA("TextButton") and element:FindFirstChild("StatusBox") ~= nil)
                                elseif searchType == "dropdown" then
                                    isMatchedType = (element:IsA("TextButton") and element:FindFirstChild("Container") ~= nil)
                                elseif searchType == "textbox" then
                                    isMatchedType = (element:IsA("Frame") and element:FindFirstChildOfClass("TextBox") ~= nil)
                                elseif searchType == "button" then
                                    isMatchedType = (element:IsA("TextButton") and element:FindFirstChild("StatusBox") == nil and element:FindFirstChild("Container") == nil)
                                end
                            end
                            
                            if isMatchedName and isMatchedType then
                                element.Visible = true
                            else
                                element.Visible = false
                            end
                        end
                    end
                end
            end
        end
    end
end

function ResetComponentSearch()
    Library.Flags["search_box_flag"] = ""
    Library.Flags["search_type_dropdown_flag"] = "Choose Type"
    pcall(function()
        if SettingTab then
            updatesaves()
        end
    end)
    
    for _, page in pairs(ContentContainer:GetChildren()) do

            for _, section in pairs(page:GetChildren()) do
                if section:IsA("ScrollingFrame") then
                    for _, element in pairs(section:GetChildren()) do
                        if element:IsA("Frame") or element:IsA("TextButton") then

                        end

        end
    end
end
